# 代码高亮样式演示

这篇文章展示了各种编程语言的代码高亮效果，以及特殊功能如行号、代码行高亮、文件名显示等。所有这些功能会根据网站的主题（明亮/暗黑）自动调整样式。

## JavaScript 示例

```js
// 简单的 JavaScript 函数
function greeting(name) {
    return `你好，${name}！`
}

// 使用箭头函数
const add = (a, b) => a + b

// 类
class Person {
    constructor(name, age) {
        this.name = name
        this.age = age
    }

    sayHello() {
        console.log(`你好，我是 ${this.name}，今年 ${this.age} 岁`)
    }
}

const person = new Person('张三', 25)
person.sayHello() // 输出: 你好，我是 张三，今年 25 岁
```

// 用户接口定义
interface User {
    id: number

    avatar?: string // 可选属性

class UserService {
    private users: User[] = []

    constructor() {
        console.log('用户服务初始化')
    }

    addUser(user: User): void {
        this.users.push(user)
    }

    getUserById(id: number): User | undefined {
        return this.users.find((user) => user.id === id)
    }
}

## Python 示例

```python title="data_processor.py"
import pandas as pd
import numpy as np
from typing import List, Dict, Any

class DataProcessor:
    """数据处理类，用于清洗和转换数据"""

    def __init__(self, data_path: str):
        """初始化数据处理器

        参数:
            data_path: 数据文件路径
        """
        self.data = pd.read_csv(data_path)
        self.processed = False

    def clean_data(self) -> None:
        """清洗数据，删除空值和异常值"""
        # 删除空值
        self.data.dropna(inplace=True)

        # 删除异常值
        q1 = self.data.quantile(0.25)
        q3 = self.data.quantile(0.75)
        iqr = q3 - q1
        self.data = self.data[~((self.data < (q1 - 1.5 * iqr)) |
                                (self.data > (q3 + 1.5 * iqr))).any(axis=1)]

        self.processed = True
        print(f"数据清洗完成，剩余 {len(self.data)} 条记录")

    def get_statistics(self) -> Dict[str, Any]:
        """获取数据统计信息

        返回:
            包含各种统计指标的字典
        """
        if not self.processed:
            print("警告: 数据尚未清洗")

        stats = {
            "count": len(self.data),
            "mean": self.data.mean().to_dict(),
            "median": self.data.median().to_dict(),
            "std": self.data.std().to_dict()
        }

        return stats
```

## SQL 示例

```sql title="create_tables.sql"
-- 创建用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP,
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

-- 创建文章表
CREATE TABLE posts (
    id SERIAL PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    content TEXT NOT NULL,
    author_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP,
    published_at TIMESTAMP,
    is_published BOOLEAN NOT NULL DEFAULT FALSE,
    slug VARCHAR(200) UNIQUE NOT NULL,
    views_count INTEGER NOT NULL DEFAULT 0
);

-- 创建评论表
CREATE TABLE comments (
    id SERIAL PRIMARY KEY,
    post_id INTEGER REFERENCES posts(id) ON DELETE CASCADE,
    user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
    parent_comment_id INTEGER REFERENCES comments(id) ON DELETE CASCADE,
    content TEXT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    is_approved BOOLEAN NOT NULL DEFAULT FALSE
);

-- 创建标签表
CREATE TABLE tags (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL,
    slug VARCHAR(50) UNIQUE NOT NULL
);

-- 创建文章-标签关联表
CREATE TABLE post_tags (
    post_id INTEGER REFERENCES posts(id) ON DELETE CASCADE,
    tag_id INTEGER REFERENCES tags(id) ON DELETE CASCADE,
    PRIMARY KEY (post_id, tag_id)
);

-- 创建索引
CREATE INDEX posts_author_id_idx ON posts(author_id);
CREATE INDEX comments_post_id_idx ON comments(post_id);
CREATE INDEX comments_user_id_idx ON comments(user_id);
```

## 特性总结

以上示例展示了我们代码高亮的主要特性：

1. **语法高亮** - 支持多种编程语言
2. **代码行号** - 自动添加行号便于参考
3. **行高亮** - 可以通过 `{行号}` 语法高亮特定行
4. **文件名显示** - 通过 `title="文件名"` 语法添加文件名
5. **复制按钮** - 每个代码块右上角的复制按钮
6. **暗色/明亮主题自适应** - 根据站点主题自动切换

试着切换网站主题，看看代码块如何无缝适应新的颜色方案！
